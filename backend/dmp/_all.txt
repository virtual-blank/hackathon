async def lifespan(app: FastAPI):
    settings = get_telegram_settings()
    if settings.webhook_url:
        await tg_client.set_webhook(settings.webhook_url)

    reminders_task = asyncio.create_task(reminders_loop())
    try:
        yield
    finally:
        reminders_task.cancel()
        with contextlib.suppress(asyncio.CancelledError):
            await reminders_task

import uuid

from sqlalchemy import and_, func, or_, select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from src.achievements.models import Achievement, RuleTypeEnum
from src.habits.models import Habit
from src.social.enums import RelationStatus
from src.social.models import UserRelation
from src.users.models import UserAchievement


class AchievementsRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def _commit(self):
        await self.session.commit()

    async def get_allowed_achievements(self, rule_value: int, rule_type: RuleTypeEnum):
        stmt = select(Achievement.id).where(Achievement.rule_type == rule_type, Achievement.rule_value == rule_value)
        result = await self.session.execute(stmt)
        return result.scalars().all()

    async def set_achievement(self, user_id: uuid.UUID, achievement_id: uuid.UUID) -> UserAchievement | None:
        new_achievement = UserAchievement(user_id=user_id, achievement_id=achievement_id)
        self.session.add(new_achievement)
        try:
            await self._commit()
            return new_achievement
        except IntegrityError:
            await self.session.rollback()
            return None

    async def get_friends_count(self, user_id: uuid.UUID):
        stmt = select(func.count(UserRelation.id)).where(
            or_(
                and_(UserRelation.user_id == user_id, UserRelation.status == RelationStatus.ACCEPTED),
                and_(UserRelation.target_id == user_id, UserRelation.status == RelationStatus.ACCEPTED),
            )
        )
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def get_habits_count(self, user_id: uuid.UUID):
        stmt = select(func.count(Habit.id)).where(Habit.creator_id == user_id) 
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

            rule_value=streak_days,

            return None

            rule_value=total_friends, rule_type=RuleTypeEnum.FRIENDS_COUNT

            return None

        for achievement_id in allowed_achievements_ids:
            new_achievement = await self.repo.set_achievement(user_id=user_id, achievement_id=achievement_id)

import uuid
from typing import Annotated

from database.db import get_db
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from sqlalchemy.ext.asyncio import AsyncSession
from src.auth.security import decode_access_token
from src.exceptions.auth_exceptions import InvalidTokenError
from src.users.models import User

SessionDep = Annotated[AsyncSession, Depends(get_db)]

bearer_scheme = HTTPBearer(auto_error=False)


async def get_current_user_id(
    credentials: Annotated[HTTPAuthorizationCredentials | None, Depends(bearer_scheme)],
) -> uuid.UUID:
    if credentials is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Не авторизован",
            headers={"WWW-Authenticate": "Bearer"},
        )
    try:
        return decode_access_token(credentials.credentials)
    except InvalidTokenError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=str(exc),
            headers={"WWW-Authenticate": "Bearer"},
        ) from exc


async def get_current_user(
    session: SessionDep,
    user_id: Annotated[uuid.UUID, Depends(get_current_user_id)],
) -> User:
    user = await session.get(User, user_id)
    if user is None:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Пользователь не найден")
    return user


CurrentUserIdDep = Annotated[uuid.UUID, Depends(get_current_user_id)]
CurrentUserDep = Annotated[User, Depends(get_current_user)]

import hashlib
import secrets
import uuid
from datetime import datetime, timedelta, timezone

import jwt
from core.config import get_jwt_settings
from src.exceptions.auth_exceptions import InvalidTokenError

ACCESS_TOKEN_EXPIRE_MINUTES = get_jwt_settings().access_token_expire_minutes
SECRET_KEY = get_jwt_settings().secret_key
ALGORITHM = get_jwt_settings().algorithm
REFRESH_TOKEN_EXPIRE_DAYS = get_jwt_settings().refresh_token_expire_days

def create_access_token(user_id: uuid.UUID, expires_delta: timedelta | None = None) -> str:
    expire = datetime.now(timezone.utc) + (expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES))
    payload = {"sub": str(user_id), "exp": expire}
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)


def decode_access_token(token: str) -> uuid.UUID:
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
    except jwt.ExpiredSignatureError as exc:
        raise InvalidTokenError("Токен истёк") from exc
    except jwt.InvalidTokenError as exc:
        raise InvalidTokenError("Невалидный токен") from exc

    sub = payload.get("sub")
    if sub is None:
        raise InvalidTokenError("Токен без sub")
    try:
        return uuid.UUID(sub)
    except ValueError as exc:
        raise InvalidTokenError("Некорректный sub в токене") from exc

def create_refresh_token() -> str:
    return secrets.token_urlsafe(64)


def hash_refresh_token(token: str) -> str:
    return hashlib.sha256(token.encode()).hexdigest()

import uuid
from datetime import date, timedelta

from sqlalchemy import and_, case, delete, exists, func, or_, select, update
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import joinedload, selectinload

from src.feed.models import Review
from src.habits.enums import Visibility
from src.habits.models import Habit, HabitLog
from src.social.enums import RelationStatus
from src.social.models import UserRelation


class FeedRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def _commit(self):
        await self.session.commit()

    async def _rollback(self):
        await self.session.rollback()

    async def get_friends_feed(self, user_id: uuid.UUID, limit: int, offset: int):
        # id всех друзей
        friends_stmt = select(
            case((UserRelation.user_id == user_id, UserRelation.target_id), else_=UserRelation.user_id)
        ).where(
            UserRelation.status == RelationStatus.ACCEPTED,
            or_(UserRelation.user_id == user_id, UserRelation.target_id == user_id),
        )

        # последняя активность, общее количество логов
        latest_activity_stmt = (
            select(
                HabitLog.habit_id,
                func.max(HabitLog.completed_date).label("last_log_date"),
                func.count(HabitLog.id).label("total_logs_count"),
            ).group_by(HabitLog.habit_id)
        ).subquery()

        ninety_days_ago = date.today() - timedelta(days=90)

        # мейн запрос 
        stmt = (
            select(
                Habit,
                func.coalesce(latest_activity_stmt.c.total_logs_count, 0).label("total_logs_count"),
            )
            .outerjoin(latest_activity_stmt, Habit.id == latest_activity_stmt.c.habit_id)
            .where(Habit.creator_id.in_(friends_stmt), Habit.visibility != Visibility.PRIVATE)
            .order_by(
                func.coalesce(latest_activity_stmt.c.last_log_date, Habit.start_date).desc(),
                Habit.created_at.desc(),
            )
            .limit(limit)
            .offset(offset)
            .options(
                joinedload(Habit.creator),
                selectinload(Habit.logs.and_(HabitLog.completed_date >= ninety_days_ago)),
                selectinload(Habit.reviews).joinedload(Review.reviewer),
            )
        )

        result = await self.session.execute(stmt)
        return [(row[0], row[1]) for row in result.unique().all()]

    async def can_user_review_habit(self, user_id: uuid.UUID, habit_id: uuid.UUID) -> None:
        habit = await self.session.get(Habit, habit_id)
        if habit is None or habit.visibility == Visibility.PRIVATE:
            raise ValueError("Привычка не найдена или скрыта")

        if habit.creator_id == user_id:
            raise ValueError("Нельзя оценивать собственную привычку")

        friend_exists_stmt = select(
            exists().where(
                UserRelation.status == RelationStatus.ACCEPTED,
                or_(
                    and_(UserRelation.user_id == user_id, UserRelation.target_id == habit.creator_id),
                    and_(UserRelation.user_id == habit.creator_id, UserRelation.target_id == user_id),
                ),
            )
        )
        is_friend = (await self.session.execute(friend_exists_stmt)).scalar()
        if not is_friend:
            raise ValueError("Оценивать можно только привычки друзей")

    async def create_review_and_get_reviewer(
        self, reviewer_id: uuid.UUID, habit_id: uuid.UUID, comment: str | None, rate: int
    ) -> Review:
        new_review = Review(reviewer_id=reviewer_id, habit_id=habit_id, comment=comment, rate=rate)
        self.session.add(new_review)
        try:
            await self._commit()
        except IntegrityError as exc:
            await self._rollback()
            raise ValueError("Вы уже оставляли реакцию на эту привычку") from exc

        stmt = select(Review).where(Review.id == new_review.id).options(joinedload(Review.reviewer))
        result = await self.session.execute(stmt)
        return result.scalar_one()

    async def edit_review(
        self, review_id: uuid.UUID, reviewer_id: uuid.UUID, new_review_dict: dict
    ) -> Review | None:
        stmt = (
            update(Review)
            .where(Review.id == review_id, Review.reviewer_id == reviewer_id)
            .values(**new_review_dict)
            .returning(Review.id)
        )
        result = await self.session.execute(stmt)
        updated_review_id = result.scalar_one_or_none()
        if not updated_review_id:
            await self._rollback()
            return None

        await self._commit()
        select_stmt = select(Review).where(Review.id == updated_review_id).options(joinedload(Review.reviewer))
        return (await self.session.execute(select_stmt)).scalar_one()

    async def delete_review(self, reviewer_id: uuid.UUID, review_id: uuid.UUID) -> bool:
        stmt = delete(Review).where(Review.id == review_id, Review.reviewer_id == reviewer_id).returning(Review.id)
        result = await self.session.execute(stmt)
        deleted_id = result.scalar_one_or_none()
        if not deleted_id:
            await self._rollback()
            return False
        await self._commit()
        return True

            habit_id=review_dto.habit_id,
            comment=review_dto.comment,
            rate=review_dto.rate,
        )

import uuid

from sqlalchemy import and_, or_, select
from sqlalchemy.ext.asyncio import AsyncSession

from src.friends.models import UserRelation
from src.social.enums import RelationStatus
from src.users.models import User


class FriendsRepository:
    """Не коммитит сам: транзакцию закрывает сервис, чтобы операция из нескольких шагов была атомарной."""

    def __init__(self, session: AsyncSession):
        self.session = session

    async def get_pair_relations(self, first_id: uuid.UUID, second_id: uuid.UUID) -> list[UserRelation]:
        """Все строки между двумя пользователями в обе стороны."""
        stmt = select(UserRelation).where(
            or_(
                and_(UserRelation.user_id == first_id, UserRelation.target_id == second_id),
                and_(UserRelation.user_id == second_id, UserRelation.target_id == first_id),
            )
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    def add_relation(self, user_id: uuid.UUID, target_id: uuid.UUID, status: RelationStatus) -> None:
        self.session.add(UserRelation(user_id=user_id, target_id=target_id, status=status))

    async def delete_relation(self, relation: UserRelation) -> None:
        await self.session.delete(relation)

    async def get_friends(self, user_id: uuid.UUID, limit: int, offset: int) -> list[User]:
        # Строка дружбы направленная, поэтому друг может стоять с любой её стороны.
        stmt = (
            select(User)
            .join(
                UserRelation,
                or_(
                    and_(UserRelation.user_id == user_id, UserRelation.target_id == User.id),
                    and_(UserRelation.target_id == user_id, UserRelation.user_id == User.id),
                ),
            )
            .where(UserRelation.status == RelationStatus.ACCEPTED)
            .order_by(User.username)
            .limit(limit)
            .offset(offset)
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def get_incoming_requests(self, user_id: uuid.UUID, limit: int) -> list[User]:
        """Пользователи, которые отправили заявку user_id; новые заявки первыми."""
        stmt = (
            select(User)
            .join(UserRelation, UserRelation.user_id == User.id)
            .where(UserRelation.target_id == user_id, UserRelation.status == RelationStatus.PENDING)
            .order_by(UserRelation.id.desc())
            .limit(limit)
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def get_outgoing_requests(self, user_id: uuid.UUID, limit: int) -> list[User]:
        """Пользователи, которым user_id отправил заявку; новые заявки первыми."""
        stmt = (
            select(User)
            .join(UserRelation, UserRelation.target_id == User.id)
            .where(UserRelation.user_id == user_id, UserRelation.status == RelationStatus.PENDING)
            .order_by(UserRelation.id.desc())
            .limit(limit)
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

import uuid

from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from src.exceptions.friends_exceptions import FriendsConflictError, FriendsNotFoundError
from src.friends.models import UserRelation
from src.friends.repository import FriendsRepository
from src.friends.schemas import FriendRequestsSchema, FriendSchema, RelationStatusSchema
from src.social.enums import RelationStatus
from src.users.repository import UsersRepository


class FriendsService:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.repo = FriendsRepository(session=session)
        self.users_repo = UsersRepository(session=session)

    async def send_request(self, user_id: uuid.UUID, target_id: uuid.UUID) -> RelationStatusSchema:
        await self._check_target(user_id=user_id, target_id=target_id)
        relations = await self.repo.get_pair_relations(user_id, target_id)

        statuses = {relation.status for relation in relations}
        if RelationStatus.BLOCKED in statuses:
            # Кто кого заблокировал, не уточняем: о чужой блокировке пользователю знать не нужно.
            raise FriendsConflictError("Нельзя отправить заявку этому пользователю")
        if RelationStatus.ACCEPTED in statuses:
            raise FriendsConflictError("Вы уже друзья")

        incoming = self._find(relations, sender_id=target_id, status=RelationStatus.PENDING)
        if incoming is not None:
            # Встречная заявка уже лежит — значит, оба согласны, сразу дружим.
            incoming.status = RelationStatus.ACCEPTED
            await self.session.commit()
            return RelationStatusSchema(status=RelationStatus.ACCEPTED)

        if self._find(relations, sender_id=user_id, status=RelationStatus.PENDING) is not None:
            raise FriendsConflictError("Заявка уже отправлена")

        self.repo.add_relation(user_id=user_id, target_id=target_id, status=RelationStatus.PENDING)
        try:
            await self.session.commit()
        except IntegrityError as exc:
            # Такой же запрос успел пройти между нашей проверкой и вставкой — сработал unique (user_id, target_id).
            await self.session.rollback()
            raise FriendsConflictError("Заявка уже отправлена") from exc
        return RelationStatusSchema(status=RelationStatus.PENDING)

    async def accept_request(self, user_id: uuid.UUID, sender_id: uuid.UUID) -> RelationStatusSchema:
        request = await self._get_incoming_request(user_id=user_id, sender_id=sender_id)
        request.status = RelationStatus.ACCEPTED
        await self.session.commit()
        return RelationStatusSchema(status=RelationStatus.ACCEPTED)

    async def decline_request(self, user_id: uuid.UUID, sender_id: uuid.UUID) -> None:
        request = await self._get_incoming_request(user_id=user_id, sender_id=sender_id)
        await self.repo.delete_relation(request)
        await self.session.commit()

    async def delete_friend(self, user_id: uuid.UUID, friend_id: uuid.UUID) -> None:
        relations = await self.repo.get_pair_relations(user_id, friend_id)
        friendship = next((relation for relation in relations if relation.status == RelationStatus.ACCEPTED), None)
        if friendship is None:
            raise FriendsNotFoundError("Этого пользователя нет в друзьях")
        await self.repo.delete_relation(friendship)
        await self.session.commit()

    async def block_user(self, user_id: uuid.UUID, target_id: uuid.UUID) -> RelationStatusSchema:
        await self._check_target(user_id=user_id, target_id=target_id)
        relations = await self.repo.get_pair_relations(user_id, target_id)

        # Заявка и дружба разрываются; встречную блокировку не трогаем — она чужая.
        for relation in relations:
            if relation.status != RelationStatus.BLOCKED:
                await self.repo.delete_relation(relation)

        if self._find(relations, sender_id=user_id, status=RelationStatus.BLOCKED) is None:
            # flush: удаление должно уйти в БД раньше вставки, иначе сработает unique (user_id, target_id).
            await self.session.flush()
            self.repo.add_relation(user_id=user_id, target_id=target_id, status=RelationStatus.BLOCKED)

        await self.session.commit()
        return RelationStatusSchema(status=RelationStatus.BLOCKED)

    async def get_friends(self, user_id: uuid.UUID, limit: int, offset: int) -> list[FriendSchema]:
        friends = await self.repo.get_friends(user_id=user_id, limit=limit, offset=offset)
        return [FriendSchema.model_validate(friend) for friend in friends]

    async def get_requests(self, user_id: uuid.UUID, limit: int) -> FriendRequestsSchema:
        incoming = await self.repo.get_incoming_requests(user_id=user_id, limit=limit)
        outgoing = await self.repo.get_outgoing_requests(user_id=user_id, limit=limit)
        return FriendRequestsSchema(
            incoming=[FriendSchema.model_validate(user) for user in incoming],
            outgoing=[FriendSchema.model_validate(user) for user in outgoing],
        )

    async def _check_target(self, user_id: uuid.UUID, target_id: uuid.UUID) -> None:
        if user_id == target_id:
            raise FriendsConflictError("Нельзя выполнить это действие с самим собой")
        if await self.users_repo.get_user_by_id(target_id) is None:
            raise FriendsNotFoundError("Пользователь не найден")

    async def _get_incoming_request(self, user_id: uuid.UUID, sender_id: uuid.UUID) -> UserRelation:
        relations = await self.repo.get_pair_relations(user_id, sender_id)
        request = self._find(relations, sender_id=sender_id, status=RelationStatus.PENDING)
        if request is None:
            raise FriendsNotFoundError("Заявка не найдена")
        return request

    @staticmethod
    def _find(relations: list[UserRelation], sender_id: uuid.UUID, status: RelationStatus) -> UserRelation | None:
        """Строка с нужным статусом, которую создал sender_id."""
        for relation in relations:
            if relation.user_id == sender_id and relation.status == status:
                return relation
        return None

    __table_args__ = (CheckConstraint("cardinality(notification_times) <= 3", name="max_3_reminders"),)


class SharedHabitUser(Uuid7PkMixin, TimeStampMixin, Base):
    """
    Партнёр по совместной привычке. Создатель сюда НЕ пишется, он уже в habits.creator_id.
    Участники = creator_id + user_id (если status = accepted).
    Партнёр отмечает выполнение в habit_logs со своим user_id.
    Совместный стрик = пересечение дат логов создателя и партнёра по этой привычке.
    Отклонить или отменить приглашение = удалить строку.
    Выход партнёра или разрыв дружбы = удалить строку и логи партнёра по этой привычке;
    привычка остаётся у создателя как личная.
    """

    __tablename__ = "shared_habit_users"

    # unique=True: в v1 только пара. Для групп замените на UniqueConstraint("habit_id", "user_id").
    habit_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("habits.id", ondelete="CASCADE"), unique=True)
    user_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"), index=True)
    status: Mapped[InviteStatus] = mapped_column(
        str_enum(InviteStatus), default=InviteStatus.PENDING, server_default=InviteStatus.PENDING.value
    )
    # Начало совместного отсчёта (для процентов). Ставит сервис при принятии.
    accepted_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))

    habit: Mapped["Habit"] = relationship(back_populates="shared_habit_user")
    user: Mapped["User"] = relationship(back_populates="shared_habit_users")


class HabitLog(Uuid7PkMixin, TimeStampMixin, Base):
    __tablename__ = "habit_logs"

    # Пользователь может отмечать привычку только если:
    # 1. он является создателем привычки;
    # 2. или он принят как партнёр в shared_habit_users.
    # Это бизнес-правило проверяется в service layer.

    habit_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("habits.id", ondelete="CASCADE"))
    user_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))

    # Календарный день в часовом поясе пользователя. БЕЗ default=date.today:
    # это дата сервера, а не пользователя. Дату считает сервис.
    completed_date: Mapped[date] = mapped_column(Date)

    habit: Mapped["Habit"] = relationship(back_populates="logs")
    user: Mapped["User"] = relationship(back_populates="habit_logs")

    __table_args__ = (
        # Одна отметка в день на человека. Покрывает и запросы «все отметки привычки».
        UniqueConstraint("habit_id", "user_id", "completed_date"),
        # История пользователя по всем привычкам за диапазон дат.
        Index("ix_habit_logs_user_date", "user_id", "completed_date"),
    )

import uuid

from typing import Any

from sqlalchemy import delete, exists, select

from src.habits.enums import InviteStatus, Visibility
from src.habits.models import Habit, HabitLog, SharedHabitUser


class HabitLogRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def mark_completed(
        self,
        *,
        user_id: uuid.UUID,
        habit_id: uuid.UUID,
        completed_date: date,
    ) -> HabitLog:
        habit = await self.session.get(Habit, habit_id)
        if habit is None:
            raise ValueError(f"Привычка {habit_id} не найдена")

        if not await self._user_can_log_habit(user_id=user_id, habit=habit):
            raise ValueError(f"Привычка {habit_id} не найдена")

        log = HabitLog(
            habit_id=habit_id,
            user_id=user_id,
            completed_date=completed_date,
        )
        self.session.add(log)
        try:
            await self.session.flush()
        except IntegrityError as exc:
            await self.session.rollback()
            raise ValueError(f"Привычка {habit_id} уже отмечена выполненной на {completed_date}") from exc

        await self.session.commit()
        await self.session.refresh(log)
        return log

    async def get_log_by_date(
        self,
        *,
        user_id: uuid.UUID,
        habit_id: uuid.UUID,
        completed_date: date,
    ) -> HabitLog | None:
        stmt = select(HabitLog).where(
            HabitLog.user_id == user_id,
            HabitLog.habit_id == habit_id,
            HabitLog.completed_date == completed_date,
        )
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def delete_log_instance(self, log: HabitLog) -> None:
        await self.session.delete(log)
        await self.session.flush()

    async def get_logs_in_range(
        self,
        *,
        user_id: uuid.UUID,
        date_from: date,
        date_to: date,
        habit_id: uuid.UUID | None = None,
        limit: int = 50,
        offset: int = 0,
    ) -> list[HabitLog]:
        if date_from > date_to:
            raise ValueError("Первая дата не может быть больше последней")

        stmt = (
            select(HabitLog)
            .where(
                HabitLog.user_id == user_id,
                HabitLog.completed_date >= date_from,
                HabitLog.completed_date <= date_to,
            )
            .order_by(HabitLog.completed_date.asc(), HabitLog.id.asc())
            .limit(limit)
            .offset(offset)
        )
        if habit_id is not None:
            stmt = stmt.where(HabitLog.habit_id == habit_id)

        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def get_completed_habits_by_date(
        self,
        *,
        user_id: uuid.UUID,
        target_date: date,
    ) -> list[Habit]:
        stmt = (
            select(Habit)
            .join(HabitLog, HabitLog.habit_id == Habit.id)
            .where(
                HabitLog.user_id == user_id,
                HabitLog.completed_date == target_date,
            )
            .order_by(Habit.title.asc())
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().unique().all())

    async def get_uncompleted_habits_by_date(
        self,
        *,
        user_id: uuid.UUID,
        target_date: date,
    ) -> list[Habit]:
        created_ids = select(Habit.id).where(
            Habit.creator_id == user_id,
            Habit.start_date <= target_date,
        )
        shared_ids = (
            select(Habit.id)
            .join(SharedHabitUser, SharedHabitUser.habit_id == Habit.id)
            .where(
                SharedHabitUser.user_id == user_id,
                SharedHabitUser.status == InviteStatus.ACCEPTED,
                Habit.start_date <= target_date,
            )
        )
        completed_subq = select(HabitLog.habit_id).where(
            HabitLog.user_id == user_id,
            HabitLog.completed_date == target_date,
        )
        stmt = (
            select(Habit)
            .where(
                Habit.id.in_(created_ids.union(shared_ids)),
                Habit.id.not_in(completed_subq),
            )
            .order_by(Habit.title.asc())
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().unique().all())

    async def get_log_dates(self, *, user_id: uuid.UUID, habit_id: uuid.UUID) -> list[date]:
        stmt = (
            select(HabitLog.completed_date)
            .where(HabitLog.user_id == user_id, HabitLog.habit_id == habit_id)
            .order_by(HabitLog.completed_date.asc())
        )
        return list((await self.session.execute(stmt)).scalars().all())

    async def _user_can_log_habit(
        self,
        *,
        user_id: uuid.UUID,
        habit: Habit,
    ) -> bool:
        """Проверить, что юзер — создатель привычки или принятый партнёр."""
        if habit.creator_id == user_id:
            return True

        stmt = select(SharedHabitUser.id).where(
            SharedHabitUser.habit_id == habit.id,
            SharedHabitUser.user_id == user_id,
            SharedHabitUser.status == InviteStatus.ACCEPTED,
        )
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none() is not None


class HabitsRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create_habit(
        self,
        creator_id: uuid.UUID,
        title: str,
        color_hex: str,
        visibility: Visibility,
        start_date: date,
        notification_times: list[time],
    ) -> Habit:
        new_habit = Habit(
            creator_id=creator_id,
            title=title,
            color_hex=color_hex,
            visibility=visibility,
            start_date=start_date,
            notification_times=notification_times,
        )
        self.session.add(new_habit)
        await self.session.commit()
        return new_habit

    async def get_habit_by_id(self, habit_id: uuid.UUID, creator_id: uuid.UUID) -> Habit | None:
        habit = select(Habit).where(Habit.id == habit_id, Habit.creator_id == creator_id)
        result = await self.session.execute(habit)
        return result.scalar_one_or_none()

    async def get_all_habits(
        self,
        creator_id: uuid.UUID,
        visibility: list[Visibility] | None = None,
        limit: int = 50,
        offset: int = 0,
        start_date: date | None = None,
        end_date: date | None = None,
        status: str | None = None,
    ) -> list[Habit]:

        if start_date is not None and end_date is not None and start_date > end_date:
            raise ValueError("Начальная дата не может быть позже конечной")

        stmt = select(Habit).where(Habit.creator_id == creator_id)

        if visibility is not None:
            stmt = stmt.where(Habit.visibility.in_(visibility))

        if end_date is not None:
            stmt = stmt.where(Habit.start_date <= end_date)

        if status is not None:
            log_exists = select(HabitLog.id).where(
                HabitLog.habit_id == Habit.id,
                HabitLog.user_id == creator_id,
            )

            if start_date is not None:
                log_exists = log_exists.where(HabitLog.completed_date >= start_date)

            if end_date is not None:
                log_exists = log_exists.where(HabitLog.completed_date <= end_date)

            condition = ~exists(log_exists) if status == "uncompleted" else exists(log_exists)
            stmt = stmt.where(condition)

        stmt = stmt.order_by(Habit.created_at.desc(), Habit.id).limit(limit).offset(offset)

        result = await self.session.execute(stmt)
        return result.scalars().all()

    async def update_habit(
        self,
        habit_id: uuid.UUID,
        creator_id: uuid.UUID,
        updates: dict[str, Any],
    ) -> Habit | None:
        habit = await self.get_habit_by_id(habit_id=habit_id, creator_id=creator_id)

        if habit is None:
            return None

        for field, value in updates.items():
            setattr(habit, field, value)

        await self.session.commit()
        await self.session.refresh(habit)
        return habit

    async def delete_habit(self, habit_id: uuid.UUID, creator_id: uuid.UUID) -> bool:
        stmt = (
            delete(Habit)
            .where(
                Habit.id == habit_id,
                Habit.creator_id == creator_id,
            )
            .returning(Habit.id)
        )
        result = await self.session.execute(stmt)
        await self.session.commit()
        return result.scalar_one_or_none() is not None

    async def update_streak(
        self,
        habit_id: uuid.UUID,
        creator_id: uuid.UUID,
        current_streak_days: int,
        max_streak_days: int,
        streak_expires_at: datetime | None,
    ) -> Habit | None:
        return await self.update_habit(
            habit_id=habit_id,
            creator_id=creator_id,
            updates={
                "current_streak_days": current_streak_days,
                "max_streak_days": max_streak_days,
                "streak_expires_at": streak_expires_at,
            },
        )

from typing import Annotated

from pydantic import BaseModel, ConfigDict, Field, field_validator, model_validator

from src.habits.enums import Visibility


class HabitLogCreateSchema(BaseModel):
    """Схема создания отметки о выполнении привычки."""

    completed_date: date | None = Field(
        default=None,
        description="Календарный день в TZ пользователя. Если None — сервис подставит сегодня.",
    )

    @field_validator("completed_date")
    @classmethod
    def not_in_future(cls, v: date | None) -> date | None:
        if v is not None and v > date.today():
            raise ValueError("Дата не может быть в будущем")
        return v


class HabitCreateSchema(BaseModel):
    model_config = ConfigDict(str_strip_whitespace=True)
    title: Annotated[str, Field(min_length=1, max_length=100)]
    color_hex: Annotated[str, Field(pattern=r"^#[0-9A-Fa-f]{6}$")] = "#4F46E5"
    visibility: Visibility = Visibility.FRIENDS
    notification_times: Annotated[list[time], Field(max_length=3)] = Field(default_factory=list)

    visibility: Visibility | None = None
    notification_times: Annotated[list[time] | None, Field(max_length=3)] = None

    model_config = ConfigDict(from_attributes=True)
    id: uuid.UUID
    title: str
    color_hex: str
    visibility: Visibility
    start_date: date
    notification_times: list[time]
    current_streak_days: int
    max_streak_days: int
    created_at: datetime
    updated_at: datetime

    @classmethod
    def from_habit(cls, habit) -> "HabitOutputSchema":
        out = cls.model_validate(habit)
        expires = habit.streak_expires_at
        if out.current_streak_days and (expires is None or expires <= datetime.now(timezone.utc)):
            out.current_streak_days = 0
        return out

def calc_streaks(dates: list[date], tz: ZoneInfo) -> tuple[int, int, datetime | None]:
    if not dates:
        return 0, 0, None
    run = best = 1
    for prev, cur in zip(dates, dates[1:]):
        run = run + 1 if (cur - prev).days == 1 else 1
        best = max(best, run)
    expires = datetime.combine(dates[-1] + timedelta(days=2), datetime.min.time(), tzinfo=tz)
    return run, best, expires

class HabitLogService:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.repo = HabitLogRepository(session=session)

    async def mark_completed(self, user_id: uuid.UUID, habit_id: uuid.UUID, completed_date: date | None = None):
        user = await self.session.get(User, user_id)
        if user is None:
            raise ValueError("Пользователь не найден")
        if not user.timezone:
            raise ValueError("Часовой пояс пользователя не указан")

        try:
            user_timezone = ZoneInfo(user.timezone)
        except ZoneInfoNotFoundError:
            raise ValueError("Некорректный часовой пояс пользователя")

        date_today = datetime.now(user_timezone).date()

        if completed_date is None:
            completed_date = date_today

        if completed_date > date_today:
            raise ValueError("Нельзя отметить привычку выполненной в будущем")

        log = await self.repo.mark_completed(user_id=user_id, habit_id=habit_id, completed_date=completed_date)
        return HabitLogOutputSchema.model_validate(log)

    async def get_logs_in_range(
        self,
        user_id: uuid.UUID,
        date_from: date,
        date_to: date,
        habit_id: uuid.UUID | None = None,
        limit: int = 50,
        offset: int = 0,
    ):
        logs = await self.repo.get_logs_in_range(
            user_id=user_id,
            date_from=date_from,
            date_to=date_to,
            habit_id=habit_id,
            limit=limit,
            offset=offset,
        )
        return [HabitLogOutputSchema.model_validate(log) for log in logs]

    async def get_completed_habits_by_date(self, user_id: uuid.UUID, target_date: date):
        completed_habits = await self.repo.get_completed_habits_by_date(user_id=user_id, target_date=target_date)
        return [HabitOutputSchema.from_habit(completed_habit) for completed_habit in completed_habits]

    async def get_uncompleted_habits_by_date(self, user_id: uuid.UUID, target_date: date):
        uncompleted_habits = await self.repo.get_uncompleted_habits_by_date(user_id=user_id, target_date=target_date)
        return [HabitOutputSchema.from_habit(uncompleted_habit) for uncompleted_habit in uncompleted_habits]


class HabitsService:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.habit_repo = HabitsRepository(session=session)
        self.habit_log_repo = HabitLogRepository(session=session)
        self.friends_repo = FriendsRepository(session=session)

    async def create_habit(self, user_id: uuid.UUID, habit_data: HabitCreateSchema):
        user = await self.session.get(User, user_id)
        if user is None:
            raise ValueError("Пользователь не найден")
        start_date = datetime.now(ZoneInfo(user.timezone)).date()
        habit = await self.habit_repo.create_habit(
            creator_id=user_id,
            title=habit_data.title,
            color_hex=habit_data.color_hex,
            visibility=habit_data.visibility,
            start_date=start_date,
            notification_times=habit_data.notification_times,
        )
        return HabitOutputSchema.from_habit(habit)

    async def get_habit(self, user_id: uuid.UUID, habit_id: uuid.UUID):
        habit = await self.habit_repo.get_habit_by_id(habit_id=habit_id, creator_id=user_id)
        if habit is None:
            return None
        return HabitOutputSchema.from_habit(habit)

    async def get_habits(
        self,
        user_id: uuid.UUID,
        visibility: list[Visibility] | None = None,
        target_user_id: uuid.UUID | None = None,
        start_date: date | None = None,
        end_date: date | None = None,
        status: str | None = None,
        limit: int = 50,
        offset: int = 0,
    ):
        if target_user_id is None or target_user_id == user_id:
            target_user_id = user_id
        else:
            relations = await self.friends_repo.get_pair_relations(first_id=user_id, second_id=target_user_id)
            is_friend = any(relation.status == RelationStatus.ACCEPTED for relation in relations)
            is_blocked = any(
                relation.status == RelationStatus.BLOCKED
                and relation.user_id == target_user_id
                and relation.target_id == user_id
                for relation in relations
            )

            if is_blocked:
                raise ValueError("Доступ к привычкам пользователя запрещён")

            if is_friend:
                visibility = [Visibility.FRIENDS, Visibility.PUBLIC]
            else:
                visibility = [Visibility.PUBLIC]

        habits = await self.habit_repo.get_all_habits(
            creator_id=target_user_id,
            visibility=visibility,
            start_date=start_date,
            end_date=end_date,
            status=status,
            limit=limit,
            offset=offset,
        )
        return [HabitOutputSchema.from_habit(habit) for habit in habits]

    async def update_habit(self, user_id: uuid.UUID, habit_id: uuid.UUID, habit_data: HabitUpdateSchema):
        updates = habit_data.model_dump(exclude_unset=True)
        if not updates:
            raise ValueError("Необходимо указать хотя бы одно поле для обновления")
        if any(value is None for value in updates.values()):
            raise ValueError("Значение None для полей не допустимо")

        habit = await self.habit_repo.update_habit(habit_id=habit_id, creator_id=user_id, updates=updates)
        if habit is None:
            return None

        return HabitOutputSchema.from_habit(habit)

    async def delete_habit(self, user_id: uuid.UUID, habit_id: uuid.UUID):
        return await self.habit_repo.delete_habit(habit_id=habit_id, creator_id=user_id)

    async def mark_habit(self, habit_id: uuid.UUID, current_user: User) -> MarkedHabitOutputSchema:
        habit = await self.session.get(Habit, habit_id)
        if not habit:
            raise ValueError("Привычка не найдена")

        if not await self.habit_log_repo._user_can_log_habit(user_id=current_user.id, habit=habit):
            raise ValueError("Привычка не найдена")

        tz = ZoneInfo(current_user.timezone or "UTC")
        today_date = datetime.now(tz).date()

        existing_log = await self.habit_log_repo.get_log_by_date(
            user_id=current_user.id, habit_id=habit_id, completed_date=today_date
        )

        if existing_log is not None:
            await self.habit_log_repo.delete_log_instance(existing_log)
            updated_habit_dao = await self._recalc_streaks(habit)
            return MarkedHabitOutputSchema(
                is_completed=False,
                log=None,
                habit=HabitOutputSchema.from_habit(updated_habit_dao),
            )

        new_log_dao = await self.habit_log_repo.mark_completed(
            user_id=current_user.id, habit_id=habit_id, completed_date=today_date
        )
        updated_habit_dao = await self._recalc_streaks(habit)
        return MarkedHabitOutputSchema(
            is_completed=True,
            log=HabitLogOutputSchema.model_validate(new_log_dao),
            habit=HabitOutputSchema.from_habit(updated_habit_dao),
)
    async def _recalc_streaks(self, habit: Habit) -> Habit:
        creator = await self.session.get(User, habit.creator_id)
        tz = ZoneInfo(creator.timezone or "UTC")
        dates = await self.habit_log_repo.get_log_dates(user_id=habit.creator_id, habit_id=habit.id)
        current, best, expires = calc_streaks(dates, tz)
        return await self.habit_repo.update_streak(
            habit_id=habit.id,
            creator_id=habit.creator_id,
            current_streak_days=current,
            max_streak_days=best,
            streak_expires_at=expires,
        )

from database.db import Base

    from src.users.models import User


class UserRelation(Uuid7PkMixin, TimeStampMixin, Base):
    """
    Дружба и блокировки в одной таблице. Строка направленная:
      PENDING  : user_id отправил заявку target_id
      ACCEPTED : дружба
      BLOCKED  : user_id заблокировал target_id

    Допустимые наборы строк для пары (A, B):
      пусто | A->B pending | A->B accepted (или B->A) | A->B blocked | B->A blocked | оба blocked
    Дружба и блокировка одновременно не существуют. Инвариант держит сервис в одной транзакции:
      - block(A, B): удалить все НЕ-blocked строки пары, затем вставить A->B blocked
        (строку B->A blocked, если она есть, НЕ трогать);
      - заявка A->B запрещена, если есть blocked в любую сторону;
      - если B->A pending уже существует, заявка A->B подтверждает дружбу.
    """
    __tablename__ = "user_relations"

    user_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))
    target_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))
    status: Mapped[RelationStatus] = mapped_column(str_enum(RelationStatus))
    user: Mapped["User"] = relationship(
        back_populates="sent_relations",
        foreign_keys=[user_id],
    )

    target: Mapped["User"] = relationship(
        back_populates="received_relations",
        foreign_keys=[target_id],
    )


    __table_args__ = (
        UniqueConstraint("user_id", "target_id"),
        CheckConstraint("user_id <> target_id", name="no_self_relation"),
        Index("ix_user_relations_target_status", "target_id", "status"),
    )


class Message(Uuid7PkMixin, TimeStampMixin, Base):
    """
    Только текст. Приглашения на совместные привычки сюда НЕ пишутся:
    они берутся из shared_habit_users со статусом pending.
    UUIDv7 растёт со временем, поэтому id годится для сортировки и курсорной пагинации.
    """
    __tablename__ = "messages"

    sender_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"))
    receiver_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("users.id", ondelete="CASCADE"), index=True)
    content: Mapped[str] = mapped_column(Text)  # длину (до 2000) проверяет Pydantic

    # Прочитано ли получателем. По умолчанию — нет.
    # Индекс нужен для быстрого запроса «непрочитанные от данного юзера».
    is_read: Mapped[bool] = mapped_column(default=False, server_default=text("false"), index=True)
    # Когда прочитано. NULL — если ещё не читали.
    read_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))

    sender: Mapped["User"] = relationship(
        back_populates="sent_messages",
        foreign_keys=[sender_id],
    )

    receiver: Mapped["User"] = relationship(
        back_populates="received_messages",
        foreign_keys=[receiver_id],
    )


    __table_args__ = (
        Index("ix_messages_dialog", "sender_id", "receiver_id", "id"),
    )

import uuid
from datetime import UTC, datetime

from sqlalchemy import and_, or_, select, update
from sqlalchemy.ext.asyncio import AsyncSession

from src.social.enums import RelationStatus
from src.social.models import Message, UserRelation


class MessagesRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create_message(
        self,
        sender_id: uuid.UUID,
        receiver_id: uuid.UUID,
        content: str,
    ) -> Message:
        """Создать новое сообщение."""
        new_message = Message(
            sender_id=sender_id,
            receiver_id=receiver_id,
            content=content,
        )
        self.session.add(new_message)
        await self.session.commit()
        await self.session.refresh(new_message)
        return new_message

    async def get_dialog(
        self,
        user1_id: uuid.UUID,
        user2_id: uuid.UUID,
        limit: int = 50,
        before_id: uuid.UUID | None = None,
    ) -> list[Message]:
        """
        История сообщений между двумя пользователями (в обе стороны).
        Сортировка — по id (UUIDv7 растёт со временем).
        Пагинация — курсорная: всё «до» before_id (для скролла вверх).
        """
        stmt = select(Message).where(
            or_(
                and_(Message.sender_id == user1_id, Message.receiver_id == user2_id),
                and_(Message.sender_id == user2_id, Message.receiver_id == user1_id),
            )
        )
        if before_id is not None:
            stmt = stmt.where(Message.id < before_id)
        stmt = stmt.order_by(Message.id.desc()).limit(limit)
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def mark_messages_read(
        self,
        message_ids: list[uuid.UUID],
        reader_id: uuid.UUID,
    ) -> list[Message]:
        """
        Пометить сообщения прочитанными.
        Только те, где receiver_id = reader_id (нельзя пометить чужие).
        """
        stmt = (
            update(Message)
            .where(
                Message.id.in_(message_ids),
                Message.receiver_id == reader_id,
                Message.is_read.is_(False),
            )
            .values(is_read=True, read_at=datetime.now(UTC))
            .returning(Message)
        )
        result = await self.session.execute(stmt)
        await self.session.commit()
        return list(result.scalars().all())

    async def get_unread(
        self,
        user_id: uuid.UUID,
        from_user_id: uuid.UUID,
    ) -> list[Message]:
        """Все непрочитанные сообщения от конкретного пользователя."""
        stmt = (
            select(Message)
            .where(
                Message.receiver_id == user_id,
                Message.sender_id == from_user_id,
                Message.is_read.is_(False),
            )
            .order_by(Message.id.asc())
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def is_blocked_between(
        self,
        user_a: uuid.UUID,
        user_b: uuid.UUID,
    ) -> bool:
        """
        Проверить, есть ли блокировка в любую сторону между двумя пользователями.
        Если A заблокировал B или B заблокировал A — сообщение запрещено.
        """
        stmt = select(UserRelation).where(
            UserRelation.status == RelationStatus.BLOCKED,
            ((UserRelation.user_id == user_a) & (UserRelation.target_id == user_b))
            | ((UserRelation.user_id == user_b) & (UserRelation.target_id == user_a)),
        ).limit(1)
        return (await self.session.execute(stmt)).first() is not None

import uuid

from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from src.social.models import Message
from src.social.repository import MessagesRepository


class SocialService:
    """Бизнес-логика чата между пользователями."""

    def __init__(self, session: AsyncSession):
        self.session = session
        self.repo = MessagesRepository(session=session)

    async def send_message(
        self,
        sender_id: uuid.UUID,
        receiver_id: uuid.UUID,
        content: str,
    ) -> Message:
        """
        Отправить сообщение.
        Проверки: не сам себе, нет блокировки, получатель существует (через FK).
        """
        if sender_id == receiver_id:
            raise ValueError("Нельзя отправить сообщение самому себе")

        if await self.repo.is_blocked_between(sender_id, receiver_id):
            raise ValueError("Невозможно отправить сообщение: пользователь недоступен")

        try:
            return await self.repo.create_message(
                sender_id=sender_id,
                receiver_id=receiver_id,
                content=content,
            )
        except IntegrityError as exc:
            if "receiver_id" in str(exc.orig):
                raise ValueError("Пользователь не найден") from exc
            raise

    async def get_history(
        self,
        user_id: uuid.UUID,
        peer_id: uuid.UUID,
        limit: int = 50,
        before_id: uuid.UUID | None = None,
    ) -> list[Message]:
        """История сообщений между двумя пользователями."""
        if user_id == peer_id:
            raise ValueError("Нельзя запросить переписку с самим собой")

        return await self.repo.get_dialog(
            user1_id=user_id,
            user2_id=peer_id,
            limit=limit,
            before_id=before_id,
        )

    async def mark_as_read(
        self,
        message_ids: list[uuid.UUID],
        reader_id: uuid.UUID,
    ) -> list[Message]:
        """Пометить сообщения прочитанными. Только полученные reader_id."""
        return await self.repo.mark_messages_read(
            message_ids=message_ids,
            reader_id=reader_id,
        )

    async def get_unread(
        self,
        user_id: uuid.UUID,
        from_user_id: uuid.UUID,
    ) -> list[Message]:

class ConnectionManager:
    """
    Реестр активных WebSocket-соединений в памяти процесса.
    user_id → множество сокетов (у юзера может быть несколько вкладок/устройств).
    """

    def __init__(self) -> None:
        self.active: dict[uuid.UUID, set[WebSocket]] = {}

    async def connect(self, user_id: uuid.UUID, websocket: WebSocket) -> None:
        await websocket.accept()
        self.active.setdefault(user_id, set()).add(websocket)

    def disconnect(self, user_id: uuid.UUID, websocket: WebSocket) -> None:
        sockets = self.active.get(user_id)
        if not sockets:
            return
        sockets.discard(websocket)
        if not sockets:
            self.active.pop(user_id, None)

    async def send_to_user(self, user_id: uuid.UUID, payload: dict) -> bool:
        """
        Отправить payload во все сокеты юзера.
        True — если доставлено (юзер онлайн), False — если офлайн.
        """
        sockets = self.active.get(user_id)
        if not sockets:
            return False
        # копия: сокет может отвалиться во время итерации
        for websocket in list(sockets):
            try:
                await websocket.send_json(payload)
            except Exception:
                self.disconnect(user_id, websocket)
        return True


manager = ConnectionManager()


@router.websocket("/ws/chat")
async def chat_websocket(websocket: WebSocket) -> None:
    """
    WebSocket endpoint для чата.
    Аутентификация — JWT в query-параметре: /ws/chat?token=<JWT>.
    """
    # == АУТЕНТИФИКАЦИЯ ==
    token = websocket.query_params.get("token")
    if not token:
        await websocket.close(code=1008, reason="Токен не передан")
        return

    try:
        user_id = decode_access_token(token)
    except InvalidTokenError as exc:
        await websocket.close(code=1008, reason=str(exc))
        return

    # == РЕГИСТРАЦИЯ В МЕНЕДЖЕРЕ ==
    await manager.connect(user_id, websocket)

    try:
        # == ЦИКЛ ОБРАБОТКИ СОБЫТИЙ ==
     while True:
         try:
             raw = await websocket.receive_json()
             if not isinstance(raw, dict):
                 raise ValueError
             await _handle_event(user_id, raw, websocket)
         except WebSocketDisconnect:
             raise
         except Exception:
             await websocket.send_json(WSErrorOut(detail="Некорректное событие").model_dump())
    except WebSocketDisconnect:
        pass
    finally:
        manager.disconnect(user_id, websocket)


async def _handle_event(user_id: uuid.UUID, raw: dict, websocket: WebSocket) -> None:
    """Роутинг события по полю `type`."""
    event_type = raw.get("type")

    if event_type == "message":
        await _handle_message(user_id, raw, websocket)
    elif event_type == "history":
        await _handle_history(user_id, raw, websocket)
    elif event_type == "mark_read":
        await _handle_mark_read(user_id, raw, websocket)
    else:
        await websocket.send_json(WSErrorOut(detail=f"Неизвестный тип события: {event_type}").model_dump())


async def _handle_message(user_id: uuid.UUID, raw: dict, websocket: WebSocket) -> None:
    """Событие `message`: сохранить и запушить получателю."""
    try:
        payload = WSMessageIn.model_validate(raw)
    except ValidationError as exc:
        await websocket.send_json(WSErrorOut(detail=str(exc)).model_dump())
        return

    async with async_session_factory() as session:
        service = SocialService(session=session)
        try:
            message = await service.send_message(
                sender_id=user_id,
                receiver_id=payload.receiver_id,
                content=payload.content,
            )
        except ValueError as exc:
            await websocket.send_json(WSErrorOut(detail=str(exc)).model_dump())
            return

    # пуш получателю, если онлайн
    await manager.send_to_user(
        payload.receiver_id,
        WSNewMessageOut(message=message).model_dump(mode="json"),
    )

    # эхо отправителю — подтверждение, что сообщение сохранено
    await websocket.send_json(WSNewMessageOut(message=message).model_dump(mode="json"))


async def _handle_history(user_id: uuid.UUID, raw: dict, websocket: WebSocket) -> None:
    """Событие `history`: отдать историю переписки с собеседником."""
    try:
        payload = WSHistoryIn.model_validate(raw)
    except ValidationError as exc:
        await websocket.send_json(WSErrorOut(detail=str(exc)).model_dump())
        return

    async with async_session_factory() as session:
        service = SocialService(session=session)
        try:
            messages = await service.get_history(
                user_id=user_id,
                peer_id=payload.peer_id,
                limit=payload.limit,
                before_id=payload.before_id,
            )
        except ValueError as exc:
            await websocket.send_json(WSErrorOut(detail=str(exc)).model_dump())
            return

    await websocket.send_json(WSHistoryOut(peer_id=payload.peer_id, messages=messages).model_dump(mode="json"))


async def _handle_mark_read(user_id: uuid.UUID, raw: dict, websocket: WebSocket) -> None:
    """Событие `mark_read`: пометить сообщения прочитанными и уведомить отправителей."""
    try:
        payload = WSMarkReadIn.model_validate(raw)
    except ValidationError as exc:
        await websocket.send_json(WSErrorOut(detail=str(exc)).model_dump())
        return

    async with async_session_factory() as session:
        service = SocialService(session=session)
        updated = await service.mark_as_read(
            message_ids=payload.message_ids,
            reader_id=user_id,
        )

    if not updated:
        return

    # уведомить каждого отправителя: его сообщения прочитаны
    message_ids_by_sender: dict[uuid.UUID, list[uuid.UUID]] = {}
    for message in updated:
        message_ids_by_sender.setdefault(message.sender_id, []).append(message.id)

    for sender_id, message_ids in message_ids_by_sender.items():
        await manager.send_to_user(
            sender_id,
            WSReadOut(by=user_id, message_ids=message_ids).model_dump(mode="json"),
        )

def _api_url(method: str) -> str:
    """Собирает URL метода Telegram Bot API."""

    return f"{TELEGRAM_API_BASE}/bot{token}/{method}"


async def send_message(chat_id: int, text: str) -> bool:
    """
    Отправляет текстовое сообщение в Telegram.
    Возвращает True, если доставлено, False — иначе.
    """
    try:
        async with httpx.AsyncClient(timeout=10.0) as client:
            response = await client.post(
                _api_url("sendMessage"),
                json={"chat_id": chat_id, "text": text},
            )
            data = response.json()
            if not data.get("ok"):
                logger.warning("Telegram sendMessage не сработал: %s", data)
                return False
            return True
    except httpx.HTTPError as exc:
        logger.error("Telegram sendMessage: ошибка HTTP: %s", exc)
        return False

    """
    Регистрирует webhook в Telegram.
    Вызывается один раз при старте приложения (если url задан).
    """
    try:
        async with httpx.AsyncClient(timeout=10.0) as client:
            response = await client.post(
                _api_url("setWebhook"),
                json={"url": url},
            )
            data = response.json()
            if not data.get("ok"):
                logger.warning("Telegram setWebhook не сработал: %s", data)
                return False
            logger.info("Telegram webhook зарегистрирован: %s", url)
            return True
    except httpx.HTTPError as exc:
        logger.error("Telegram setWebhook: ошибка HTTP: %s", exc)
        return False

async def _handle_start(text: str, chat_id: int, session: AsyncSession) -> None:
    parts = text.split(maxsplit=1)

    if len(parts) == 1:
        await client.send_message(
            chat_id=chat_id,
            text=(
                "👋 Привет! Я бот-помощник для трекинга привычек.\n\n"

                "(кнопка «Привязать Telegram»)."
            ),
        )
        return

    code = parts[1].strip()
    if not code:
        await client.send_message(chat_id=chat_id, text="❌ Пустой код привязки")
        return

    service = TelegramService(session=session)
    try:
        await service.verify_start_code(code=code, chat_id=chat_id)

        logger.info("Привязка отклонена для chat_id=%s: %s", chat_id, exc)

        logger.exception("НЕОЖИДАННАЯ ОШИБКА при привязке chat_id=%s", chat_id)
        await client.send_message(chat_id=chat_id, text="❌ Внутренняя ошибка")

    await client.send_message(
        chat_id=chat_id,
        text="✅ Telegram привязан! Теперь ты будешь получать напоминания о привычках.",

import asyncio
import logging
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo

from sqlalchemy import func, select

from database.db import async_session_factory
from src.habits.models import Habit, HabitLog
from src.telegram.service import TelegramService
from src.users.models import User

logger = logging.getLogger(__name__)

CHECK_INTERVAL_SEC = 60
WINDOW = timedelta(minutes=30)  


def _due_slot(habit: Habit, tz: ZoneInfo, now_utc: datetime) -> datetime | None:
    now_local = now_utc.astimezone(tz)
    due = []
    for t in habit.notification_times:
        slot = datetime.combine(now_local.date(), t, tzinfo=tz)
        if (
            slot <= now_local
            and now_local - slot <= WINDOW
            and slot >= habit.created_at  # слот до создания привычки (не напоминаем)
            and (habit.last_reminded_at is None or habit.last_reminded_at < slot)
        ):
            due.append(slot)
    return max(due) if due else None


async def _tick() -> None:
    now_utc = datetime.now(timezone.utc)
    async with async_session_factory() as session:
        rows = (
            await session.execute(
                select(Habit, User.timezone, User.telegram_chat_id)
                .join(User, User.id == Habit.creator_id)
                .where(
                    User.telegram_chat_id.is_not(None),
                    User.email.not_like("%@seed.test"), 
                    func.cardinality(Habit.notification_times) > 0,
                )
            )
        ).all()

        service = TelegramService(session=session)
        for habit, tz_name, chat_id in rows:
            try:
                tz = ZoneInfo(tz_name or "UTC")
            except Exception:
                tz = ZoneInfo("UTC")

            slot = _due_slot(habit, tz, now_utc)
            if slot is None:
                continue

            done = await session.scalar(
                select(HabitLog.id)
                .where(
                    HabitLog.habit_id == habit.id,
                    HabitLog.user_id == habit.creator_id,
                    HabitLog.completed_date == slot.date(),
                )
                .limit(1)
            )
            if done:
                continue

            if await service.send_reminder(chat_id=chat_id, habit_title=habit.title):
                habit.last_reminded_at = now_utc

        await session.commit()


async def reminders_loop() -> None:
    while True:
        try:
            await _tick()
        except asyncio.CancelledError:
            raise
        except Exception:
            logger.exception("Ошибка в цикле напоминаний")
        await asyncio.sleep(CHECK_INTERVAL_SEC)

import hashlib
import logging
import secrets
import uuid
from datetime import datetime, timedelta, timezone

from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from core.config import get_telegram_settings
from src.telegram import client
from src.users.models import User
from src.users.repository import UsersRepository

logger = logging.getLogger(__name__)


class TelegramService:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.repo = UsersRepository(session=session)

    async def generate_link(self, user: User) -> str:

        settings = get_telegram_settings()
        code = secrets.token_urlsafe(32)
        code_hash = hashlib.sha256(code.encode()).hexdigest()

        
        expires_at = datetime.now(timezone.utc) + timedelta(
            minutes=settings.link_ttl_minutes
        )

        
        user.telegram_link_token_hash = code_hash
        user.telegram_link_expires_at = expires_at
        await self.session.commit()

        url = f"https://t.me/{settings.bot_name}?start={code}"
        logger.info("Сгенерирована ссылка привязки для юзера %s", user.id)
        return url

    async def verify_start_code(self, code: str, chat_id: int) -> None:

        code_hash = hashlib.sha256(code.encode()).hexdigest()

        
        user = await self.repo.get_user_by_telegram_hash(token_hash=code_hash)
        if user is None:
            raise ValueError("Ссылка недействительна")

        if user.telegram_link_expires_at is None or user.telegram_link_expires_at < datetime.now(timezone.utc):
            raise ValueError("Ссылка устарела, сгенерируйте новую")

        existing = await self.repo.get_user_by_telegram_chat_id(chat_id=chat_id)
        if existing is not None and existing.id != user.id:
            raise ValueError("Этот Telegram уже привязан к другому аккаунту")

       
        user.telegram_chat_id = chat_id
        user.telegram_link_token_hash = None
        user.telegram_link_expires_at = None

        try:
            await self.session.commit()
        except IntegrityError as exc:
            await self.session.rollback()
            raise ValueError("Не удалось привязать Telegram, попробуйте позже") from exc

        logger.info("Юзер %s привязал Telegram chat_id=%s", user.id, chat_id)

    async def send_reminder(self, chat_id: int, habit_title: str) -> bool:
        text = f"⏰ Напоминание: привычка «{habit_title}» ещё не выполнена"
        ok = await client.send_message(chat_id=chat_id, text=text)
        if not ok:
            logger.warning(
                "Не удалось отправить напоминание chat_id=%s habit=%s",
                chat_id, habit_title,
            )
        return ok

import logging
import uuid

import aioboto3
import botocore
from botocore.exceptions import BotoCoreError, ClientError
from uuid6 import uuid7

from core.config import get_minio_settings

logger = logging.getLogger(__name__)

# ClientError — хранилище ответило отказом, BotoCoreError — до него не достучались.
STORAGE_ERRORS = (ClientError, BotoCoreError)
NOT_FOUND_CODE = "404"

MINIO_ENDPOINT = get_minio_settings().endpoint
BUCKET_NAME = get_minio_settings().bucket_name
ACCESS_KEY = get_minio_settings().root_user
SECRET_KEY = get_minio_settings().root_password

PUBLIC_PREFIX = f"{MINIO_ENDPOINT}/{BUCKET_NAME}/"
DEFAULT_PFP_URL = f"{PUBLIC_PREFIX}defaults/default-pfp.webp"
ACHIEVEMENTS_PREFIX = f"{MINIO_ENDPOINT}/achievements/"


class MinioService:
    def __init__(self):
        self.session = aioboto3.Session(
            aws_access_key_id=ACCESS_KEY,
            aws_secret_access_key=SECRET_KEY,
            region_name="us-east-1",
        )
        self.endpoint_url = MINIO_ENDPOINT
        self.bucket_name = BUCKET_NAME

    def _get_client(self):
        config = botocore.client.Config(signature_version="s3v4")
        return self.session.client("s3", endpoint_url=self.endpoint_url, config=config)

    async def generate_presigned_upload_url(self, user_id: uuid.UUID, extension: str = "webp") -> dict[str, str]:
        file_key = f"pfp/{user_id}/{uuid7().hex}.{extension}"

        # Ссылка подписывается локально, без запроса в хранилище, поэтому ошибки здесь
        # не ловим: если она случилась, это неверная настройка, и её должно быть видно.
        async with self._get_client() as s3_client:
            presigned_url = await s3_client.generate_presigned_url(
                ClientMethod="put_object",
                Params={"Bucket": self.bucket_name, "Key": file_key, "ContentType": f"image/{extension}"},
                ExpiresIn=get_minio_settings().ttl_link_sec,
            )
        public_url = f"{PUBLIC_PREFIX}{file_key}"
        return {"upload_url": presigned_url, "file_key": file_key, "public_url": public_url}

    async def find_object_size(self, file_key: str) -> int | None:
        """Размер загруженного файла в байтах или None, если по ключу ничего не лежит."""
        async with self._get_client() as s3_client:
            try:
                response = await s3_client.head_object(Bucket=self.bucket_name, Key=file_key)
            except ClientError as exc:
                if exc.response["Error"]["Code"] == NOT_FOUND_CODE:
                    return None
                raise
        return response["ContentLength"]

    async def delete_avatar(self, file_key: str) -> bool:
        """Не бросает: вызывается после записи в БД, и сбой хранилища не должен ронять запрос."""
        try:
            async with self._get_client() as s3_client:
                await s3_client.delete_object(Bucket=self.bucket_name, Key=file_key)
        except STORAGE_ERRORS:
            logger.exception("pfp: не удалось удалить файл %s", file_key)
            return False
        return True


minio_service = MinioService()

@router.put("")
async def update_pfp(data: PfpUpdateRequest, user: CurrentUserDep, session: SessionDep):
    if not data.file_key.startswith(f"pfp/{user.id}/"):
        raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Чужой файл")

    new_url = f"{PUBLIC_PREFIX}{data.file_key}"
    old_url = user.pfp_url

    if old_url == new_url:
        return {"pfp_url": new_url}

    # Клиент мог получить ссылку и ничего не загрузить — без проверки в профиль
    # попала бы ссылка на несуществующий файл.
    size = await minio_service.find_object_size(data.file_key)
    if size is None:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Файл не загружен")
    if size > MAX_PFP_SIZE_BYTES:
        # Presigned PUT не умеет ограничивать размер, поэтому проверяем после загрузки.
        await minio_service.delete_avatar(data.file_key)
        raise HTTPException(status_code=status.HTTP_413_CONTENT_TOO_LARGE, detail="Файл больше 5 МБ")

    user.pfp_url = new_url
    await session.commit()

    old_key = old_url.removeprefix(PUBLIC_PREFIX) if old_url else None
    if old_key and old_key.startswith("pfp/"):
        await minio_service.delete_avatar(old_key)

    return {"pfp_url": new_url}


# == DELETE PFP ==
@router.delete("", status_code=status.HTTP_204_NO_CONTENT)
async def delete_pfp(user: CurrentUserDep, session: SessionDep) -> None:
    old_url = user.pfp_url
    if old_url is None:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Фото профиля не установлено")

    # Сначала БД, потом файл: при сбое между шагами останется лишний файл,
    # а не профиль со ссылкой на удалённый.
    user.pfp_url = None
    await session.commit()

    old_key = old_url.removeprefix(PUBLIC_PREFIX)
    if old_key.startswith("pfp/"):
        await minio_service.delete_avatar(old_key)

import uuid
from datetime import UTC, datetime

from sqlalchemy import delete, select, update
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload
from sqlalchemy import and_, delete, or_, select, update
from src.social.enums import RelationStatus
from src.social.models import UserRelation
from src.users.models import RefreshToken, User, UserAchievement


class UsersRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def _commit(self):
        await self.session.commit()

    async def _rollback(self):
        await self.session.rollback()

    async def create_user(self, username: str, password_hash: str, email: str) -> User:
        new_user = User(username=username, password_hash=password_hash, email=email, achievements=[])
        self.session.add(new_user)
        await self.session.commit()
        return new_user

    async def get_user_by_id(self, user_id: uuid.UUID) -> User | None:
        stmt = (
            select(User)
            .where(User.id == user_id)
            .options(selectinload(User.achievements).joinedload(UserAchievement.achievement))
        )
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def get_user_by_username(self, username: str) -> User | None:
        stmt = (
            select(User)
            .where(User.username == username)
            .options(selectinload(User.achievements).joinedload(UserAchievement.achievement))
        )

        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def get_user_by_telegram_hash(self, token_hash: str) -> User | None:
        stmt = select(User).where(User.telegram_link_token_hash == token_hash)
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def get_user_by_telegram_chat_id(self, chat_id: int) -> User | None:
        stmt = select(User).where(User.telegram_chat_id == chat_id)
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def delete_user(self, user_id: uuid.UUID) -> bool:
        stmt = delete(User).where(User.id == user_id)
        result = await self.session.execute(stmt)
        await self.session.commit()
        return result.rowcount > 0

    async def create_refresh_token(
        self,
        user_id: uuid.UUID,
        token_hash: str,
        expires_at: datetime,
    ) -> RefreshToken:
        new_token = RefreshToken(
            user_id=user_id,
            token_hash=token_hash,
            expires_at=expires_at,
        )
        self.session.add(new_token)
        await self.session.commit()
        return new_token

    async def get_refresh_token_by_hash(self, token_hash: str) -> RefreshToken | None:
        stmt = select(RefreshToken).where(RefreshToken.token_hash == token_hash)
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def revoke_refresh_token(self, token_id: uuid.UUID) -> None:
        stmt = update(RefreshToken).where(RefreshToken.id == token_id).values(revoked_at=datetime.now(UTC))
        await self.session.execute(stmt)
        await self.session.commit()

    async def revoke_all_user_refresh_tokens(self, user_id: uuid.UUID) -> None:
        """Отозвать все refresh юзера — при подозрении на кражу или logout."""
        stmt = (
            update(RefreshToken)
            .where(
                RefreshToken.user_id == user_id,
                RefreshToken.revoked_at.is_(None),
            )
            .values(revoked_at=datetime.now(UTC))
        )
        await self.session.execute(stmt)
        await self.session.commit()

    async def edit_profile(
        self,
        user_id: uuid.UUID,
        user_data_dict: dict,
    ):
        stmt = (
            update(User)
            .where(User.id == user_id)
            .values(**user_data_dict)
            .returning(User)
            
        )
        result = await self.session.execute(stmt)
        await self._commit()
        return result.scalar_one_or_none()

    async def search_users(self, query: str, exclude_user_id: uuid.UUID, limit: int) -> list[User]:
        # экранирование
        escaped = query.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")
        blocked = select(UserRelation.id).where(
            UserRelation.status == RelationStatus.BLOCKED,
            or_(
                and_(UserRelation.user_id == exclude_user_id, UserRelation.target_id == User.id),
                and_(UserRelation.target_id == exclude_user_id, UserRelation.user_id == User.id),
            ),
        )
        stmt = (
            select(User)
            .where(
                User.id != exclude_user_id,
                User.username.ilike(f"%{escaped}%", escape="\\"),
                ~blocked.exists(),
            )
            .order_by(User.username)
            .limit(limit)
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    def extract_achievements(cls, v: Any) -> list[Any]:
        if not v:
            return []
        return [
            item.achievement if hasattr(item, "achievement") else item
            for item in v
            if getattr(item, "achievement", item) is not None
        ]

import uuid

    create_access_token,
    create_refresh_token,
    hash_refresh_token,

class UsersService:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.repo = UsersRepository(session=session)

    async def _issue_tokens(self, user: User) -> AuthOutputSchema:
        """Выдаёт access + refresh, сохраняет refresh в БД."""
        access_str = create_access_token(user_id=user.id)

        refresh_str = create_refresh_token()
        refresh_hash = hash_refresh_token(refresh_str)
        expires_at = datetime.now(UTC) + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)

        await self.repo.create_refresh_token(
            user_id=user.id,
            token_hash=refresh_hash,
            expires_at=expires_at,
        )

        return AuthOutputSchema(
            user=UserOutputSchema.model_validate(user),
            token=TokenSchema(access_token=access_str, refresh_token=refresh_str),
        )

    async def sign_up(self, user_data: UserCreateSchema):
        password = user_data.password.get_secret_value()
        password_hash = await create_password(password=password)
        try:
            new_user_dao = await self.repo.create_user(
                username=user_data.username,
                password_hash=password_hash,
                email=user_data.email,
            )
        except IntegrityError as e:
            if "username" in str(e.orig):
                raise UsernameTakenError("Этот юзернейм занят")
            if "email" in str(e.orig):
                raise UsernameTakenError("Этот email уже зарегистрирован")
            raise

        return await self._issue_tokens(new_user_dao)

    async def login(self, user_data: UserLoginSchema):
        password = user_data.password.get_secret_value()
        user_dao = await self.repo.get_user_by_username(username=user_data.username)
        if not user_dao:
            await verify_password(password=password, hashed_password=DUMMY_HASH)
            raise IncorrectPassword("Неверный логин или пароль")
        is_password_correct = await verify_password(password=password, hashed_password=user_dao.password_hash)
        if not is_password_correct:
            raise IncorrectPassword("Неверный логин или пароль")

        return await self._issue_tokens(user_dao)

    async def refresh(self, refresh_token_str: str) -> AuthOutputSchema:
        """Обновляет access по refresh. Ротирует refresh."""
        token_hash = hash_refresh_token(refresh_token_str)
        token_dao = await self.repo.get_refresh_token_by_hash(token_hash=token_hash)

        if not token_dao:
            raise AuthError("Невалидный refresh-токен")

        if token_dao.revoked_at is not None:
            if datetime.now(UTC) - token_dao.revoked_at > timedelta(seconds=30):
                await self.repo.revoke_all_user_refresh_tokens(user_id=token_dao.user_id)
            raise AuthError("Refresh-токен отозван")

        if token_dao.expires_at < datetime.now(UTC):
            raise AuthError("Refresh-токен истёк")

        user_dao = await self.repo.get_user_by_id(user_id=token_dao.user_id)
        if not user_dao:
            raise UserNotFoundError("Пользователь не найден")

        # Ротация: старый отзываем, выдаём новую пару
        await self.repo.revoke_refresh_token(token_id=token_dao.id)
        return await self._issue_tokens(user_dao)

    async def logout(self, user_id: uuid.UUID) -> None:
        """Отзывает все refresh юзера. Access умрёт сам через exp."""
        await self.repo.revoke_all_user_refresh_tokens(user_id=user_id)

    async def delete_account(self, user_id: uuid.UUID) -> None:
        deleted = await self.repo.delete_user(user_id=user_id)
        if not deleted:
            raise UserNotFoundError("Пользователь не найден")

    async def get_me(self, user_id: uuid.UUID):
        user_dao = await self.repo.get_user_by_id(user_id=user_id)
        return UserOutputSchema.model_validate(user_dao)

    async def edit_profile(self, user_id: uuid.UUID, user_data: UserUpdateSchema):
        user_data_dict = user_data.model_dump(exclude_unset=True, exclude_none=True)
        if not user_data_dict:
            raise DataFormatError("Данные не могут быть пустые")
        try:
            updated_user = await self.repo.edit_profile(user_id=user_id, user_data_dict=user_data_dict)
            if not updated_user:
                raise UserNotFoundError("Не удалось внести изменения")
            user = await self.repo.get_user_by_id(user_id=updated_user.id)
            return UserOutputSchema.model_validate(user)

        except IntegrityError as exc:
            exc_text = str(exc.orig)
            await self.repo._rollback()
            if "username" in exc_text:
                raise UsernameTakenError("Этот юзернейм занят")
            if "email" in exc_text:
                raise UsernameTakenError("Этот имейл уже кем-то используется")
            if "timezone" in exc_text:
                raise DataFormatError("Часовой пояс указан не верно")
            if "theme" in exc_text:
                raise DataFormatError("Оформление указано не верно")
            else:
                raise

    async def change_password(self, user_id: uuid.UUID, new_password_dto: UserPasswordUpdateSchema):
        current_password = new_password_dto.current_password.get_secret_value()
        new_password = new_password_dto.new_password.get_secret_value()

        new_password_hash = await create_password(password=new_password)

        if not new_password_hash:
            raise IncorrectPassword("Не удалось проверить пароли")

        user_dao = await self.repo.get_user_by_id(user_id=user_id)
        if not user_dao:
            raise UserNotFoundError("Пользователь не найден")

        is_password_correct = await verify_password(password=current_password, hashed_password=user_dao.password_hash)
        if not is_password_correct:
            raise IncorrectPassword("Текущий пароль не верный")

        user_dao.password_hash = new_password_hash
        await self.repo._commit()

        return await self._issue_tokens(user_dao)

    async def search_users(self, current_user_id: uuid.UUID, query: str, limit: int) -> list[UserSearchSchema]:
        users = await self.repo.search_users(
            query=query.strip().lower(), exclude_user_id=current_user_id, limit=limit
        )
        return [UserSearchSchema.model_validate(user) for user in users]
